    \textbf{DANH MỤC TỪ VIẾT TẮT} \\
\end{center}
\vspace{0.5cm}

\begin{longtable}{l p{5cm} p{6cm}}
\hline
\textbf{Viết tắt} & \textbf{Tên tiếng Anh} & \textbf{Tên tiếng Việt} \\ \hline
\textbf{ACS} & Auto Configuration Server & Máy chủ tự động cấu hình thiết bị \\
\textbf{CPE} & Customer Premises Equipment & Thiết bị đầu cuối tại phía thuê bao \\
\textbf{CVE} & Common Vulnerabilities and Exposures & Mã định danh lỗ hổng phổ biến \\
\textbf{CVSS} & Common Vulnerability Scoring System & Hệ thống chấm điểm lỗ hổng \\
\textbf{DREAD} & Damage, Reproducibility, Exploitability, Affected users, Discoverability & Mô hình chấm điểm rủi ro theo thiệt hại, khả năng tái hiện, khả năng khai thác, người dùng bị ảnh hưởng và khả năng phát hiện \\
\textbf{EPSS} & Exploit Prediction Scoring System & Hệ thống dự đoán khả năng bị khai thác \\
\textbf{IoT} & Internet of Things & Internet vạn vật \\
\textbf{IRT} & Incident Response Team & Đội phản ứng sự cố \\
\textbf{NOC} & Network Operations Center & Trung tâm điều hành mạng \\
\textbf{ONT} & Optical Network Terminal & Thiết bị đầu cuối mạng quang \\
\textbf{OTA} & Over-the-Air & Cập nhật phần mềm qua mạng \\
\textbf{PASTA} & Process for Attack Simulation and Threat Analysis & Quy trình mô phỏng tấn công và phân tích mối đe dọa \\
\textbf{SBOM} & Software Bill of Materials & Bảng kê thành phần phần mềm \\
\textbf{SIEM} & Security Information and Event Management & Quản lý thông tin và sự kiện an ninh \\
\textbf{SOC} & Security Operations Center & Trung tâm điều hành an ninh \\
\textbf{STRIDE} & Spoofing, Tampering, Repudiation, Information disclosure, Denial of service, Elevation of privilege & Mô hình phân loại mối đe dọa theo sáu nhóm giả mạo, sửa đổi, chối bỏ, lộ thông tin, từ chối dịch vụ và leo thang đặc quyền \\
\textbf{TEE} & Trusted Execution Environment & Môi trường thực thi tin cậy \\
\textbf{USP} & User Services Platform & Nền tảng quản trị thiết bị theo TR-369 \\
\hline
\end{longtable}
